// /tests/app/shell-redesign-0531.test.cjs
'use strict';
const { test } = require('node:test');
const assert = require('node:assert/strict');
const fs = require('node:fs');
const path = require('node:path');
const read = (f) => fs.readFileSync(path.join(__dirname, '..', '..', f), 'utf8');

test('shell: sidebar on wide screens (collapsible), bottom tabs on phones, one project selector in the toolbar', () => {
  const app = read('src/app/ModernApp.js');
  assert.match(app, /<Sidebar [^>]*collapsed=\{collapsed\} onToggle=\{toggleSidebar\}/);
  assert.match(app, /<MobileHeader /);
  assert.match(app, /<TabBar tabs=\{TABS\}/);
  assert.match(app, /ContentWidthProvider/);
  const bars = read('src/components/shell/Bars.js');
  assert.match(bars, /Switch project/);
  assert.match(bars, /ProjectPanel/);
  assert.match(read('src/components/shell/Menus.js'), /Expand sidebar/);
});
test('sidebar choice is stored under its own key and nothing else is touched', () => {
  const src = read('src/components/shell/useSidebar.js');
  assert.match(src, /niyarvo-projects-sidebar-v1/);
  assert.doesNotMatch(src, /niyarvo-projects-v1|niyarvo-workspace/);
});
test('web: no rubber-band scroll, no tap delay, text selection still limited to inputs', () => {
  const app = read('src/app/ModernApp.js');
  assert.match(app, /overscroll-behavior:none/);
  assert.match(app, /touch-action:manipulation/);
  assert.match(app, /input,textarea\{-webkit-user-select:text/);
});
test('design system: no serif face, no uppercase labels, no emoji glyphs in the UI kit and screens', () => {
  const files = [];
  const walk = (d) => fs.readdirSync(path.join(__dirname, '..', '..', d), { withFileTypes: true }).forEach((e) => {
    const rel = `${d}/${e.name}`;
    if (e.isDirectory()) walk(rel); else if (/\.js$/.test(e.name)) files.push(rel);
  });
  ['src/ui', 'src/screens', 'src/components/shell', 'src/components/inline'].forEach(walk);
  for (const f of files) {
    const src = read(f);
    assert.doesNotMatch(src, /serifFace|textTransform: ?'uppercase'/, f);
    assert.doesNotMatch(src, /[\u{1F300}-\u{1FAFF}]/u, f);
  }
});
test('Kanban keeps its drag contract while columns adapt to the width', () => {
  const board = read('src/screens/board/Board.js');
  assert.match(board, /layout\.boardColumnWidth\(bp, avail, lists\.length, GAP/);
  assert.match(board, /step=\{colW \+ GAP\}/);
  assert.match(board, /useHoldDrag/);
});
test('Timeline: toolbar, month band, weekend columns and neutral dependency lines', () => {
  const t = read('src/screens/timeline/Timeline.js');
  assert.match(t, /Segmented options=\{\['Days', 'Weeks', 'Months'\]\}/);
  assert.match(t, /visibleMonths\.map/);
  assert.match(t, /weekendDays\.map/);
  assert.doesNotMatch(t, /LINK_COLORS/);
  assert.match(t, /useState\(wide\)/);
});
test('Priority and status are never colour only on cards and rows', () => {
  assert.match(read('src/screens/board/Board.js'), /accessibilityLabel=\{`Priority \$\{t\.priority\}`\}/);
  assert.match(read('src/ui/uiBase.js'), /<T size=\{12\} weight="500" muted numberOfLines=\{1\}>\{label\}<\/T>/);
});

test('every file that renders <T> imports it (parse checks cannot see a missing import)', () => {
  const walk = (d, out = []) => { for (const e of fs.readdirSync(path.join(__dirname, '..', '..', d), { withFileTypes: true })) { const rel = `${d}/${e.name}`; if (e.isDirectory()) walk(rel, out); else if (/\.js$/.test(e.name)) out.push(rel); } return out; };
  for (const f of [...walk('src'), 'App.js']) {
    const src = read(f);
    if (!/<T[\s>]/.test(src)) continue;
    assert.match(src, /import\s*\{[^}]*\bT\b[^}]*\}\s*from|function T\b|const T\b/, `${f} renders <T> without importing it`);
  }
});
test('filled buttons use the contrast-checked accent fill, accent text uses accentText', () => {
  assert.match(read('src/ui/uiControls.js'), /primary \? c\.accentFill/);
  assert.match(read('src/ui/uiControls.js'), /solid \? c\.accentFill/);
  assert.doesNotMatch(read('src/ui/uiControls.js'), /primary \? c\.accent :/);
});
